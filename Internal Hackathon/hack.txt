        if (netStatusDiv && netText) {

      window.addEventListener('offline', updateNetworkStatus);
      updateNetworkStatus();

    // Offline Memory Buffer Sync Function
    function syncOfflineMemoryBuffer() {
      try {
        const localData = localStorage.getItem('ks_offline_passes');
        if (localData) {
          const parsed = JSON.parse(localData);
          validBookedPasses = { ...validBookedPasses, ...parsed };
        } else {
          localStorage.setItem('ks_offline_passes', JSON.stringify(validBookedPasses));
        }

        const localBookedSlots = localStorage.getItem('ks_booked_slots');
        if (localBookedSlots) {
          bookedSlotsLedger = JSON.parse(localBookedSlots);
        } else {
          localStorage.setItem('ks_booked_slots', JSON.stringify(bookedSlotsLedger));
        }
      } catch(e) {
        console.warn("Offline storage buffer restricted or unavailable.", e);
      }
    }

    function savePassToOfflineBuffer(passCode, passObject) {
      validBookedPasses[passCode] = passObject;
      try {
        localStorage.setItem('ks_offline_passes', JSON.stringify(validBookedPasses));
      } catch(e) {
        console.warn("Could not persist to offline memory buffer.", e);
      }
    }

    // 1-Hour Hold Timer (Continuous Timer)
    function startRedisLockTimer(lockKey) {
      if (currentLockKey !== lockKey) {
        if (redisLockTimerInterval) clearInterval(redisLockTimerInterval);
        currentLockKey = lockKey;
        lockDurationRemaining = 3600; // Reset to 1 hour (3600 seconds) only when switching to a NEW unbooked slot key
      } else if (redisLockTimerInterval) {
        return; // Keep existing active countdown running without resetting
      }

      lockedSlotsLedger[lockKey] = Date.now() + (lockDurationRemaining * 1000);
      const countdownEl = document.getElementById('lock-countdown');
      
      redisLockTimerInterval = setInterval(() => {
        let minutes = parseInt(lockDurationRemaining / 60, 10);
        let seconds = parseInt(lockDurationRemaining % 60, 10);

        minutes = minutes < 10 ? "0" + minutes : minutes;
        seconds = seconds < 10 ? "0" + seconds : seconds;

        if (countdownEl) countdownEl.textContent = minutes + ":" + seconds;

        if (--lockDurationRemaining < 0) {
          clearInterval(redisLockTimerInterval);
          redisLockTimerInterval = null;
          delete lockedSlotsLedger[lockKey];
          currentLockKey = null;
          closeBookingModal();
        }
      }, 1000);
    }

    function updateTotalPrice() {
      if (!currentSelectedFacility) return;
      let slotRate = currentSelectedFacility.rate;
      let depositFee = currentSelectedFacility.deposit || 0;
      let equipFee = 0;

      const checkedEquipment = document.querySelectorAll('input[name="equipment-addon"]:checked');
      checkedEquipment.forEach(item => {
        equipFee += parseInt(item.value);
      });

      let total = slotRate + equipFee + depositFee;

      document.getElementById('summary-slot-fee').innerText = `₹${slotRate}`;
      document.getElementById('summary-equip-fee').innerText = `₹${equipFee}`;
      document.getElementById('summary-deposit-fee').innerText = `₹${depositFee}`;
      document.getElementById('booking-total-price').innerText = `₹${total}.00 Total`;
    }

    // Custom Timing Option Toggle Handler with Double-Booking Check and Continuous Timer support
    function toggleCustomTiming(val) {
      const customContainer = document.getElementById('custom-timing-container');
      const startInput = document.getElementById('custom-start-time');
      const endInput = document.getElementById('custom-end-time');

      if (val === 'CUSTOM') {
        customContainer.classList.remove('hidden');
        startInput.required = true;
        endInput.required = true;
      } else {
        customContainer.classList.add('hidden');
        startInput.required = false;
        endInput.required = false;

        // Check availability for selected pre-set slot and alert if booked
        if (currentSelectedFacility) {
          if (isSlotBooked(currentSelectedFacility.id, val)) {
            showSlotBookedModal();
            // Timer continues as it is; do NOT restart it or modify the current timer duration
          } else {
            const lockKey = `${currentSelectedFacility.id}_${val}`;
            startRedisLockTimer(lockKey);
          }
        }
      }
    }

    // Format 24hr Time String to 12hr AM/PM format
    function formatTo12Hr(timeStr) {
      if (!timeStr) return "";
      let [hours, minutes] = timeStr.split(':');
      hours = parseInt(hours);
      const ampm = hours >= 12 ? 'PM' : 'AM';
      hours = hours % 12 || 12;
      return `${hours}:${minutes} ${ampm}`;
    }

    // Dynamic Rolling QR Generator Implementation
    function startDynamicRollingQR(basePassCode) {
      activeRollingPassCode = basePassCode;
      
      if (rollingQrInterval) clearInterval(rollingQrInterval);
      if (rollingCountdownInterval) clearInterval(rollingCountdownInterval);

      const renderQR = () => {
        const rollingSeed = Math.floor(Date.now() / 15000);
        const dynamicPayload = `${basePassCode}_ROLL_${rollingSeed}`;

        const qrContainer = document.getElementById('qrcode');
        qrContainer.innerHTML = '';
        
        new QRCode(qrContainer, {
          text: dynamicPayload,
          width: 140,
          height: 140
        });
      };

      renderQR();

      let secondsLeft = 15;
      const countdownEl = document.getElementById('qr-refresh-timer');

      rollingCountdownInterval = setInterval(() => {
        secondsLeft--;
        if (secondsLeft <= 0) secondsLeft = 15;
        if (countdownEl) countdownEl.innerText = secondsLeft;
      }, 1000);

      rollingQrInterval = setInterval(() => {
        renderQR();
      }, 15000);
    }

    function startCameraScanner() {
      const qrContainer = document.getElementById('qr-reader');
      qrContainer.classList.remove('hidden');
      document.getElementById('btn-start-scanner').classList.add('hidden');
      document.getElementById('btn-stop-scanner').classList.remove('hidden');

      html5QrScanner.start(
        { facingMode: "environment" },
        { fps: 10, qrbox: { width: 220, height: 220 } },
        (decodedText) => {
          stopCameraScanner();
          processPassVerification(decodedText);
        },
        (errorMessage) => {}
      ).catch(err => {
        stopCameraScanner();
      });
    }

    // Universal File Upload QR Scanner
    function scanQrFromFile(event) {
      const file = event.target.files[0];
      if (!file) return;

      const html5QrCode = new Html5Qrcode("qr-reader");
      html5QrCode.scanFile(file, true)
        .then(decodedText => {
          processPassVerification(decodedText);
        })
        .catch(err => {
          showInvalidQrModal("Could not read a valid QR code image.");
        });
    }

    function verifyPass() {
      const passId = document.getElementById('manual-pass-id').value.trim();
      if (!passId) return;
      processPassVerification(passId);
    }

    // Pass Verification Logic with Offline Fallback & Institute Approval Checking
    function processPassVerification(rawScannedCode) {
      const resultDiv = document.getElementById('scan-result');
      resultDiv.classList.remove('hidden');

      let passCode = rawScannedCode;
      let isValidRollingQR = true;

      if (rawScannedCode.includes('_ROLL_')) {
        const parts = rawScannedCode.split('_ROLL_');
        passCode = parts[0];
        const scannedSeed = parseInt(parts[1]);
        const currentSeed = Math.floor(Date.now() / 15000);

        if (Math.abs(currentSeed - scannedSeed) > 1) {
          isValidRollingQR = false;
        }
      }

      // Check in-memory ledger or sync from localStorage offline buffer
      let passInfo = validBookedPasses[passCode];
      if (!passInfo) {
        syncOfflineMemoryBuffer();
        passInfo = validBookedPasses[passCode];
      }

      const isOffline = !navigator.onLine;

      if (passInfo && isValidRollingQR) {
        // Verification based on approval status set by admin
        const isApproved = (passInfo.approvalStatus === "Approved" || !passInfo.approvalStatus);

        if (isApproved) {
          resultDiv.className = "p-4 rounded-lg border bg-emerald-50 border-emerald-300 text-emerald-900 space-y-1.5";
          resultDiv.innerHTML = `
            <div class="flex items-center font-bold text-base text-emerald-800">
              <i data-lucide="check-circle-2" class="w-5 h-5 mr-2 text-emerald-600"></i> ENTRY GRANTED - VALID PASS ${isOffline ? '(OFFLINE MODE)' : ''}
            </div>
            <div><strong>Athlete Name:</strong> ${passInfo.name}</div>
            <div><strong>Institute Ground:</strong> ${passInfo.facility}</div>
            <div><strong>Institute Approval:</strong> <span class="font-bold text-emerald-700">APPROVED</span></div>
            <div><strong>Reserved Slot:</strong> ${passInfo.slot}</div>
            <div><strong>Team Size:</strong> ${passInfo.players || 1} Players</div>
            <div><strong>Equipment Rented:</strong> ${passInfo.equipment || 'None'}</div>
            <div><strong>Refundable Deposit:</strong> ₹${passInfo.deposit || 0}</div>
            <div><strong>Pass Code:</strong> ${passCode}</div>
            <div class="text-[10px] text-emerald-700 mt-1"> Verified via Local Offline Memory Buffer.</div>
          `;
        } else if (passInfo.approvalStatus === "Pending") {
          resultDiv.className = "p-4 rounded-lg border bg-amber-50 border-amber-300 text-amber-900 space-y-1.5";
          resultDiv.innerHTML = `
            <div class="flex items-center font-bold text-base text-amber-800">
              <i data-lucide="clock" class="w-5 h-5 mr-2 text-amber-600"></i> ENTRY PENDING - AWAITING INSTITUTE APPROVAL
            </div>
            <div><strong>Athlete Name:</strong> ${passInfo.name}</div>
            <div><strong>Institute Approval Status:</strong> <span class="font-bold text-amber-700">Pending Review</span></div>
            <div>Please contact institute administration to confirm your entry permission.</div>
          `;
        } else {
          resultDiv.className = "p-4 rounded-lg border bg-rose-50 border-rose-300 text-rose-900 space-y-1.5";
          resultDiv.innerHTML = `
            <div class="flex items-center font-bold text-base text-rose-800">
              <i data-lucide="x-circle" class="w-5 h-5 mr-2 text-rose-600"></i> ENTRY REJECTED - INSTITUTE DISAPPROVED
            </div>
            <div><strong>Athlete Name:</strong> ${passInfo.name}</div>
            <div><strong>Institute Approval Status:</strong> <span class="font-bold text-rose-700">Rejected</span></div>
            <div>Campus entry has been denied by the institute administration.</div>
          `;
        }
      } else if (passInfo && !isValidRollingQR) {
        resultDiv.className = "p-4 rounded-lg border bg-rose-50 border-rose-300 text-rose-900 space-y-1";
        resultDiv.innerHTML = `
          <div class="flex items-center font-bold text-base text-rose-800">
            <i data-lucide="x-circle" class="w-5 h-5 mr-2 text-rose-600"></i> ENTRY DENIED - EXPIRED / SCREENSHOT DETECTED
          </div>
          <div>Dynamic Rolling QR token expired or screenshot shared. Please use live pass view.</div>
        `;
        showInvalidQrModal(`Screenshot detected or rolling QR token expired for pass ${passCode}.`);
      } else {
        resultDiv.className = "p-4 rounded-lg border bg-rose-50 border-rose-300 text-rose-900 space-y-1";
        resultDiv.innerHTML = `
          <div class="flex items-center font-bold text-base text-rose-800">
            <i data-lucide="x-circle" class="w-5 h-5 mr-2 text-rose-600"></i> ENTRY DENIED - INVALID PASS
          </div>
          <div>Code <strong>${rawScannedCode}</strong> is not recognized in active booking ledger.</div>
        `;
        showInvalidQrModal(`Code "${rawScannedCode}" does not match active records.`);
      }
      lucide.createIcons();
    }
